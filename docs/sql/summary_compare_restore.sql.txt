-- Snapshot / compare / restore the About text (seo_summary) and Key Takeaways (key_takeaways) for ONE video.
-- Run on the TEST database only. Replace 123 with the video's internal ID (videos.id) in every statement.
-- The watch page's About tab reads video_transcripts.seo_summary and its Key Takeaways read video_transcripts.key_takeaways.
-- The regenerate action overwrites both and keeps no copy of the old values, so snapshot first.

-- A. BEFORE regenerating: snapshot (also copies the things that must NOT change, for the check in C)
CREATE TABLE IF NOT EXISTS summary_test_snapshot (
  video_id int NOT NULL PRIMARY KEY, taken_at datetime NOT NULL,
  seo_summary mediumtext, key_takeaways json,
  description mediumtext, plain_text_md5 char(32), plain_text_len int, vtt_path varchar(500), article_md5 char(32)
);
REPLACE INTO summary_test_snapshot
SELECT t.video_id, NOW(), t.seo_summary, t.key_takeaways, v.description, MD5(t.plain_text), CHAR_LENGTH(t.plain_text), t.vtt_path,
       (SELECT MD5(CONCAT_WS('|', a.raw_transcript, a.cleaned_transcript, a.final_article_markdown, a.status)) FROM video_seo_articles a WHERE a.video_id = t.video_id)
FROM video_transcripts t JOIN videos v ON v.id = t.video_id WHERE t.video_id = 123;

-- B. AFTER regenerating: old vs new, side by side
SELECT 'OLD' AS version, s.seo_summary AS about_text, s.key_takeaways FROM summary_test_snapshot s WHERE s.video_id = 123
UNION ALL
SELECT 'NEW', t.seo_summary, t.key_takeaways FROM video_transcripts t WHERE t.video_id = 123;

-- C. Everything else must be unchanged: every column below should read 'same'
SELECT IF(v.description <=> s.description, 'same', 'CHANGED') AS description,
       IF(MD5(t.plain_text) = s.plain_text_md5, 'same', 'CHANGED') AS transcript,
       IF(t.vtt_path <=> s.vtt_path, 'same', 'CHANGED') AS caption_path,
       IF((SELECT MD5(CONCAT_WS('|', a.raw_transcript, a.cleaned_transcript, a.final_article_markdown, a.status)) FROM video_seo_articles a WHERE a.video_id = t.video_id) <=> s.article_md5, 'same', 'CHANGED') AS seo_article
FROM video_transcripts t JOIN videos v ON v.id = t.video_id JOIN summary_test_snapshot s ON s.video_id = t.video_id WHERE t.video_id = 123;

-- D. RESTORE the old About + Key Takeaways (only if the new ones are not good enough)
UPDATE video_transcripts t JOIN summary_test_snapshot s ON s.video_id = t.video_id
SET t.seo_summary = s.seo_summary, t.key_takeaways = s.key_takeaways, t.updated_at = UNIX_TIMESTAMP()
WHERE t.video_id = 123;

-- E. When finished (after you keep or restore): DROP TABLE summary_test_snapshot;
